**信号排名算法**:

```python
def rank_signals(buy_signals, regime):
    ranked = []
    for s in buy_signals:
        # 基础分 = 信号强度 × 10
        score = s.strength * 10

        # 市场匹配加分
        if regime == 'strong' and s.strategy_style == 'trend':
            score += 0.3     # 趋势策略在强势市场加分
        elif regime != 'strong' and s.strategy_style == 'reversion':
            score += 0.3     # 回归策略在非强势市场加分

        ranked.append(score, s)

    # 交叉确认加分
    code_counts = count_by_code(ranked)
    for r in ranked:
        if code_counts[r.code] >= 2:
            r.score += 0.5    # 同一股票多策略选中
            r.cross = True    # 标记 ⭐⭐

    return sorted(ranked, key=lambda x: -x.score)
```

**SVG K线图渲染算法**:

```python
def render_svg_kline(ohlc_data, ma20, ma60):
    """生成 120 天 OHLC K 线 SVG"""

    w, h = 800, 260          # 画布尺寸
    margin = {'top': 20, 'right': 20, 'bottom': 30, 'left': 50}
    plot_w = w - margin['left'] - margin['right']
    plot_h = h - margin['top'] - margin['bottom']

    # 计算价格范围
    all_prices = [d.high for d in ohlc_data] + [d.low for d in ohlc_data]
    price_min, price_max = min(all_prices), max(all_prices)
    price_range = price_max - price_min or 1

    # 每根 K 线宽度
    bar_w = max(1, plot_w / len(ohlc_data) - 1)

    svg_parts = [f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {w} {h}">']

    # 网格线
    for i in range(5):
        y = margin['top'] + plot_h * i / 4
        price = price_max - price_range * i / 4
        svg_parts.append(f'<line x1="{margin["left"]}" y1="{y:.0f}" '
            f'x2="{w-margin["right"]}" y2="{y:.0f}" '
            f'stroke="var(--border)" stroke-width="0.5"/>')
        svg_parts.append(f'<text x="{margin["left"]-5}" y="{y+4:.0f}" '
            f'text-anchor="end" fill="var(--text-soft)" font-size="9">'
            f'{price:.1f}</text>')

    # K线
    for i, d in enumerate(ohlc_data):
        x = margin['left'] + i * plot_w / len(ohlc_data)
        is_up = d.close >= d.open
        color = 'var(--up)' if is_up else 'var(--down)'

        # 影线
        high_y = margin['top'] + (price_max - d.high) / price_range * plot_h
        low_y = margin['top'] + (price_max - d.low) / price_range * plot_h
        svg_parts.append(f'<line x1="{x+bar_w/2:.1f}" y1="{high_y:.1f}" '
            f'x2="{x+bar_w/2:.1f}" y2="{low_y:.1f}" stroke="{color}" stroke-width="1"/>')

        # 实体
        body_top = margin['top'] + (price_max - max(d.open, d.close)) / price_range * plot_h
        body_h = max(1, abs(d.close - d.open) / price_range * plot_h)
        svg_parts.append(f'<rect x="{x:.1f}" y="{body_top:.1f}" '
            f'width="{bar_w:.1f}" height="{body_h:.1f}" fill="{color}"/>')

    # MA20 线
    if ma20:
        pts = []
        for i, v in enumerate(ma20[-len(ohlc_data):]):
            if v is None: continue
```

## 四、信号排名算法

```python
def compute_signal_ranking(signals, regime):
    """
    输入: signal_history 中最新一天 status='passed' 的 BUY 信号
          market regime ('strong' | 'weak')

    输出: 排名列表 [{rank, code, name, strategies[], price, score, cross, reason}]
    """

    ranked = []
    for sig in signals:
        # === 1. 基础分 ===
        base = sig['strength'] * 10  # 信号强度映射到 0-1 区间

        # === 2. 市场匹配加分 ===
        regime_bonus = 0.0
        style = get_strategy_style(sig['strategy'])
        if regime == 'strong' and style == 'trend':
            regime_bonus = 0.3
        elif regime != 'strong' and style == 'reversion':
            regime_bonus = 0.3

        # === 3. 价格可买性 ===
        affordable = sig['price'] <= 50

        ranked.append({
            'code': sig['code'],
            'name': sig['name'],
            'strategy': sig['strategy'],
            'price': sig['price'],
            'score': base + regime_bonus,
            'base': base,
            'regime_bonus': regime_bonus,
            'affordable': affordable,
            'reason': sig['reason'],
        })

    # === 4. 合并同股票多策略 ===
    merged = {}
    for r in ranked:
        code = r['code']
        if code not in merged:
            merged[code] = {**r, 'strategies': [r['strategy']], 'max_score': r['score']}
        else:
            merged[code]['strategies'].append(r['strategy'])
            merged[code]['max_score'] = max(merged[code]['max_score'], r['score'])

    # === 5. 交叉确认加分 ===
    for code, m in merged.items():
        if len(m['strategies']) >= 2:
            m['max_score'] += 0.5
            m['cross'] = True
        else:
            m['cross'] = False

    # === 6. 过滤 + 排序 ===
    affordable_only = [m for m in merged.values() if m['affordable']]
    affordable_only.sort(key=lambda x: -x['max_score'])

    return affordable_only
```
